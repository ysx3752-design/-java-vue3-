create table device_status
(
    id           int auto_increment comment '状态ID'
        primary key,
    device_type  varchar(50)                          not null comment '设备类型',
    status       tinyint(1) default 0                 null comment '设备状态',
    last_updated datetime   default CURRENT_TIMESTAMP null on update CURRENT_TIMESTAMP comment '最后更新时间',
    description  varchar(100)                         null comment '设备描述'
)
    comment '设备状态表';

create table operation_logs
(
    id               int auto_increment comment '日志ID'
        primary key,
    user_id          int                                null comment '操作用户ID',
    operation_type   varchar(50)                        not null comment '操作类型',
    device_type      varchar(50)                        null comment '设备类型',
    operation_time   datetime default CURRENT_TIMESTAMP null comment '操作时间',
    operation_result tinyint(1)                         null comment '操作结果',
    description      text                               null comment '操作描述'
)
    comment '操作日志表';

create table sensor_data
(
    id                int auto_increment comment '数据ID'
        primary key,
    timestamp         datetime default CURRENT_TIMESTAMP null comment '记录时间',
    air_temperature   float                              null comment '空气温度(°C)',
    air_humidity      float                              null comment '空气湿度(%)',
    soil_temperature  float                              null comment '土壤温度(°C)',
    soil_moisture     float                              null comment '土壤湿度(%)',
    light_intensity   float                              null comment '光照强度(lux)',
    co2_concentration float                              null comment 'CO2浓度(ppm)'
)
    comment '传感器数据表';

create table users
(
    id            int auto_increment comment '用户ID'
        primary key,
    username      varchar(80)                           not null comment '用户名',
    password_hash varchar(256)                          not null comment '密码哈希',
    email         varchar(120)                          not null comment '邮箱',
    is_active     tinyint(1)  default 1                 null comment '是否激活',
    created_at    datetime    default CURRENT_TIMESTAMP null comment '创建时间',
    role          varchar(20) default 'user'            null comment '用户角色',
    constraint email
        unique (email),
    constraint username
        unique (username)
)
    comment '用户信息表';

create table threshold_settings
(
    id          int auto_increment comment '规则ID'
        primary key,
    name        varchar(50)                          not null comment '规则名称',
    type        varchar(20)                          not null comment '监测类型',
    operator    varchar(5)                           not null comment '触发条件操作符',
    threshold   float                                not null comment '阈值',
    enabled     tinyint(1) default 1                 null comment '是否启用',
    create_time datetime   default CURRENT_TIMESTAMP null comment '创建时间',
    update_time datetime   default CURRENT_TIMESTAMP null on update CURRENT_TIMESTAMP comment '更新时间',
    user_id     int                                  null comment '创建者ID',
    constraint threshold_settings_ibfk_1
        foreign key (user_id) references users (id)
)
    comment '阈值设置表';
 
create index user_id
    on threshold_settings (user_id);
